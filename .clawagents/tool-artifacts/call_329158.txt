 175:     async def _on_tts(self, websocket: WebSocketServerProtocol, payload: dict[str, Any]) -> None:
 176:         request_id = str(payload.get('request_id', ''))
 177:         text = str(payload.get('text', '')).strip()
 178:         if not text:
 179:             await websocket.send(json.dumps({'type': 'tts.error', 'request_id': request_id, 'message': 'No text to speak.'}))
 180:             return
 181:         try:
 182:             self._load_kemetone()

 185:             encoded = base64.b64encode(audio.tobytes()).decode('ascii')
 186:             await websocket.send(json.dumps({
 187:                 'type': 'tts.audio',

 189:                 'audio': encoded,

 191:                 'sample_rate': 24000
 192:             }))
 193:         except Exception as exc:
